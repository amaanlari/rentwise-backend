// title
title Rentwise Data Model

// define tables
owners [icon: user, color: yellow]{
  id string pk
  name string
  email string unique
  phone_number string unique
  password string

  created_at date
  updated_at date
}

rooms [icon: home, color: blue]{
  id string pk
  owner_id string
  room_number string
  room_type enum
  current_rent_amount decimal
  security_deposit decimal
  status enum
  notes string

  created_at date
  updated_at date
}

tenants [icon: user-check, color: green]{
  id string pk
  room_id string
  name string
  email string unique
  phone_number string unique
  id_proof_number string
  emergency_contact_name string
  emergency_contact_number string
  joining_date date
  exit_date date
  lease_start_date date
  lease_end_date date (null)
  notes string

  created_at date
  updated_at date
}

rents [icon: dollar-sign, color: orange]{
  id string pk
  room_id string unique
  month int unique
  year int unique
  amount decimal
  status string
  due_date date
  paid_date date
  notes string

  created_at date
  updated_at date
}
